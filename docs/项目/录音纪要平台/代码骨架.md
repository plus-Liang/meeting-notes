## 3. `docker-compose.yml` 骨架

```yaml
services:
  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes: ["pgdata:/var/lib/postgresql/data"]
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
      interval: 5s
      retries: 20

  redis:
    image: redis:7-alpine
    volumes: ["redisdata:/data"]
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      retries: 20

  minio:
    image: minio/minio:latest
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: ${MINIO_ACCESS_KEY}
      MINIO_ROOT_PASSWORD: ${MINIO_SECRET_KEY}
    ports: ["9000:9000", "9001:9001"]
    volumes: ["miniodata:/data"]
    healthcheck:
      test: ["CMD", "mc", "ready", "local"]      # 或 curl 健康端点
      interval: 5s
      retries: 20

  asr:
    image: ${ASR_IMAGE}          # FunASR 官方镜像 或 自建 Dockerfile
    # 暴露 OpenAI 兼容接口，内部端口以镜像文档为准
    ports: ["${ASR_PORT}:${ASR_PORT}"]
    healthcheck:
      test: ["CMD-SHELL", "curl -fsS http://localhost:${ASR_PORT}/v1/models || exit 1"]
      interval: 10s
      retries: 30

  app:
    build: ./app
    env_file: [.env]
    depends_on:
      postgres: {condition: service_healthy}
      redis: {condition: service_healthy}
      minio: {condition: service_healthy}
      asr: {condition: service_healthy}
    ports: ["8080:8080"]

volumes:
  pgdata: {}
  redisdata: {}
  miniodata: {}
```

> **不要用 `sleep` 等依赖**，一律 healthcheck + `condition: service_healthy`。

## 4. `resources/db/migration/V1__init.sql`

```sql
CREATE TABLE task (
  id            BIGSERIAL PRIMARY KEY,
  user_id       BIGINT       NOT NULL DEFAULT 1,
  source_type   VARCHAR(16)  NOT NULL,          -- UPLOAD/FETCH/PASTE/RECORD
  source_uri    TEXT,
  content_hash  VARCHAR(64)  NOT NULL,          -- 幂等键 (D-009 / N1)
  duration_sec  INT,
  size_bytes    BIGINT,
  status        VARCHAR(16)  NOT NULL,          -- PENDING/RUNNING/DONE/RETRYING/DEAD
  attempts      INT          NOT NULL DEFAULT 0,
  error_code    VARCHAR(64),
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ  NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX ux_task_hash   ON task(content_hash);   -- 幂等来源
CREATE INDEX        ix_task_status ON task(status, updated_at);

CREATE TABLE task_result (
  task_id     BIGINT PRIMARY KEY REFERENCES task(id) ON DELETE CASCADE,
  transcript  TEXT,
  summary     TEXT,
  todos       JSONB,
  keywords    JSONB,
  speaker_map JSONB,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE glossary (
  id         BIGSERIAL PRIMARY KEY,
  user_id    BIGINT      NOT NULL DEFAULT 1,
  wrong      VARCHAR(128) NOT NULL,
  right      VARCHAR(128) NOT NULL,
  hits       INT         NOT NULL DEFAULT 1,
  priority   INT         NOT NULL DEFAULT 0,
  UNIQUE (user_id, wrong)
);

CREATE TABLE speaker_profile (
  id        BIGSERIAL PRIMARY KEY,
  user_id   BIGINT      NOT NULL DEFAULT 1,
  label     VARCHAR(64) NOT NULL,
  note      TEXT
);

CREATE TABLE quota (
  user_id      BIGINT NOT NULL,
  day          DATE   NOT NULL,
  used_minutes INT    NOT NULL DEFAULT 0,
  used_tasks   INT    NOT NULL DEFAULT 0,
  PRIMARY KEY (user_id, day)
);

CREATE TABLE dead_letter (
  id         BIGSERIAL PRIMARY KEY,
  task_id    BIGINT REFERENCES task(id),
  reason     TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
```

## 8. 编码顺序提醒

1. 先让 `PASTE` 入口通（不依赖任何模型），能最快跑通"输入→结构化→输出"，
   并且它天然验证了 LLM 通道与 JSON Schema。
2. 再接 `UPLOAD`，此时只需在中间插一个转写步骤。
3. 最后接 `FETCH`（安全校验最多，放最后）。

> 顺序本身也是一个面试点：**"我先做不依赖外部模型的最短闭环，把架构风险前置暴露。"**
